# Time Credit Exchange System - Database Setup

## Overview
The Time Credit Exchange System enables users to request, accept, and complete exchanges of time credits. This is the core functionality that makes FutureBank a functional time banking platform.

## Database Schema

### 1. Exchanges Table
```sql
-- Create exchanges table to track exchange requests
CREATE TABLE public.exchanges (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  
  -- Exchange details
  offer_id BIGINT NOT NULL REFERENCES public.offers(id) ON DELETE CASCADE,
  want_id BIGINT NOT NULL REFERENCES public.wants(id) ON DELETE CASCADE,
  
  -- Participants
  provider_id UUID NOT NULL REFERENCES public.profiles(id) ON DELETE CASCADE,
  requester_id UUID NOT NULL REFERENCES public.profiles(id) ON DELETE CASCADE,
  
  -- Exchange status
  status TEXT NOT NULL DEFAULT 'pending' CHECK (status IN ('pending', 'accepted', 'completed', 'cancelled', 'declined')),
  
  -- Time credit amount
  time_credits DECIMAL(5,2) NOT NULL CHECK (time_credits > 0),
  
  -- Exchange details
  scheduled_date TIMESTAMP WITH TIME ZONE,
  location TEXT,
  notes TEXT,
  
  -- Completion details
  completed_at TIMESTAMP WITH TIME ZONE,
  completion_notes TEXT,
  
  -- Cancellation details
  cancelled_at TIMESTAMP WITH TIME ZONE,
  cancellation_reason TEXT,
  cancelled_by UUID REFERENCES public.profiles(id)
);

-- Add indexes for performance
CREATE INDEX idx_exchanges_provider_id ON public.exchanges(provider_id);
CREATE INDEX idx_exchanges_requester_id ON public.exchanges(requester_id);
CREATE INDEX idx_exchanges_status ON public.exchanges(status);
CREATE INDEX idx_exchanges_offer_id ON public.exchanges(offer_id);
CREATE INDEX idx_exchanges_want_id ON public.exchanges(want_id);
CREATE INDEX idx_exchanges_created_at ON public.exchanges(created_at);

-- Add updated_at trigger
CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = NOW();
    RETURN NEW;
END;
$$ language 'plpgsql';

CREATE TRIGGER update_exchanges_updated_at 
    BEFORE UPDATE ON public.exchanges 
    FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();
```

### 2. Transactions Table
```sql
-- Create transactions table to track time credit transfers
CREATE TABLE public.transactions (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  
  -- Transaction details
  exchange_id BIGINT REFERENCES public.exchanges(id) ON DELETE SET NULL,
  transaction_type TEXT NOT NULL CHECK (transaction_type IN ('exchange', 'donation', 'withdrawal', 'refund')),
  
  -- Participants
  from_user_id UUID REFERENCES public.profiles(id) ON DELETE SET NULL,
  to_user_id UUID REFERENCES public.profiles(id) ON DELETE SET NULL,
  
  -- Amount
  amount DECIMAL(5,2) NOT NULL,
  
  -- Balance snapshots
  from_balance_before DECIMAL(5,2),
  from_balance_after DECIMAL(5,2),
  to_balance_before DECIMAL(5,2),
  to_balance_after DECIMAL(5,2),
  
  -- Metadata
  description TEXT,
  metadata JSONB DEFAULT '{}'::jsonb
);

-- Add indexes
CREATE INDEX idx_transactions_from_user_id ON public.transactions(from_user_id);
CREATE INDEX idx_transactions_to_user_id ON public.transactions(to_user_id);
CREATE INDEX idx_transactions_exchange_id ON public.transactions(exchange_id);
CREATE INDEX idx_transactions_created_at ON public.transactions(created_at);
CREATE INDEX idx_transactions_type ON public.transactions(transaction_type);
```

### 3. Enable RLS and Create Policies

```sql
-- Enable RLS
ALTER TABLE public.exchanges ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.transactions ENABLE ROW LEVEL SECURITY;

-- Exchange policies
CREATE POLICY "Users can view exchanges they're involved in" ON public.exchanges
  FOR SELECT USING (
    auth.uid() = provider_id OR 
    auth.uid() = requester_id
  );

CREATE POLICY "Users can create exchange requests" ON public.exchanges
  FOR INSERT WITH CHECK (
    auth.uid() = requester_id AND
    auth.uid() != provider_id
  );

CREATE POLICY "Users can update exchanges they're involved in" ON public.exchanges
  FOR UPDATE USING (
    auth.uid() = provider_id OR 
    auth.uid() = requester_id
  );

-- Transaction policies
CREATE POLICY "Users can view their own transactions" ON public.transactions
  FOR SELECT USING (
    auth.uid() = from_user_id OR 
    auth.uid() = to_user_id
  );

CREATE POLICY "System can create transactions" ON public.transactions
  FOR INSERT WITH CHECK (true); -- Transactions are created by system functions
```

### 4. Exchange Management Functions

```sql
-- Function to create an exchange request
CREATE OR REPLACE FUNCTION create_exchange_request(
  p_offer_id BIGINT,
  p_want_id BIGINT,
  p_time_credits DECIMAL(5,2),
  p_scheduled_date TIMESTAMP WITH TIME ZONE DEFAULT NULL,
  p_location TEXT DEFAULT NULL,
  p_notes TEXT DEFAULT NULL
)
RETURNS BIGINT AS $$
DECLARE
  v_exchange_id BIGINT;
  v_offer_user_id UUID;
  v_want_user_id UUID;
BEGIN
  -- Get the user IDs
  SELECT user_id INTO v_offer_user_id FROM public.offers WHERE id = p_offer_id;
  SELECT user_id INTO v_want_user_id FROM public.wants WHERE id = p_want_id;
  
  -- Validate that the current user is the want owner
  IF auth.uid() != v_want_user_id THEN
    RAISE EXCEPTION 'You can only create exchange requests for your own wants';
  END IF;
  
  -- Validate that the user isn't requesting their own offer
  IF v_offer_user_id = v_want_user_id THEN
    RAISE EXCEPTION 'You cannot request your own offer';
  END IF;
  
  -- Check if there's already a pending exchange for this offer/want combination
  IF EXISTS (
    SELECT 1 FROM public.exchanges 
    WHERE offer_id = p_offer_id 
    AND want_id = p_want_id 
    AND status = 'pending'
  ) THEN
    RAISE EXCEPTION 'An exchange request already exists for this offer and want';
  END IF;
  
  -- Create the exchange
  INSERT INTO public.exchanges (
    offer_id, want_id, provider_id, requester_id, 
    time_credits, scheduled_date, location, notes
  ) VALUES (
    p_offer_id, p_want_id, v_offer_user_id, v_want_user_id,
    p_time_credits, p_scheduled_date, p_location, p_notes
  ) RETURNING id INTO v_exchange_id;
  
  RETURN v_exchange_id;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Function to accept an exchange request
CREATE OR REPLACE FUNCTION accept_exchange_request(p_exchange_id BIGINT)
RETURNS VOID AS $$
DECLARE
  v_exchange RECORD;
BEGIN
  -- Get the exchange details
  SELECT * INTO v_exchange FROM public.exchanges WHERE id = p_exchange_id;
  
  -- Validate that the current user is the provider
  IF auth.uid() != v_exchange.provider_id THEN
    RAISE EXCEPTION 'Only the provider can accept exchange requests';
  END IF;
  
  -- Validate the exchange status
  IF v_exchange.status != 'pending' THEN
    RAISE EXCEPTION 'Only pending exchanges can be accepted';
  END IF;
  
  -- Update the exchange status
  UPDATE public.exchanges 
  SET status = 'accepted', updated_at = NOW()
  WHERE id = p_exchange_id;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Function to decline an exchange request
CREATE OR REPLACE FUNCTION decline_exchange_request(
  p_exchange_id BIGINT,
  p_reason TEXT DEFAULT NULL
)
RETURNS VOID AS $$
DECLARE

BEGIN

  -- Validate that the current user is the provider
  IF auth.uid() != v_exchange.provider_id THEN

  IF v_exchange.status != 'pending' THEN

  SET status = 'declined', updated_at = NOW()

$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Function to complete an exchange
CREATE OR REPLACE FUNCTION complete_exchange(
  p_exchange_id BIGINT,
  p_completion_notes TEXT DEFAULT NULL
)
RETURNS VOID AS $$
DECLARE
  v_exchange RECORD;
  v_provider_balance DECIMAL(5,2);
  v_requester_balance DECIMAL(5,2);
BEGIN
  -- Get the exchange details
  SELECT * INTO v_exchange FROM public.exchanges WHERE id = p_exchange_id;
  
  -- Validate that the current user is the provider
  IF auth.uid() != v_exchange.provider_id THEN
    RAISE EXCEPTION 'Only the provider can complete exchanges';
  END IF;
  
  -- Validate the exchange status
  IF v_exchange.status != 'accepted' THEN
    RAISE EXCEPTION 'Only accepted exchanges can be completed';
  END IF;
  
  -- Get current balances
  SELECT time_balance INTO v_provider_balance FROM public.profiles WHERE id = v_exchange.provider_id;
  SELECT time_balance INTO v_requester_balance FROM public.profiles WHERE id = v_exchange.requester_id;
  
  -- Validate requester has enough credits
  IF v_requester_balance < v_exchange.time_credits THEN
    RAISE EXCEPTION 'Requester does not have enough time credits to complete this exchange';
  END IF;
  
  -- Start transaction
  BEGIN
    -- Update exchange status
    UPDATE public.exchanges 
    SET status = 'completed', 
        completed_at = NOW(), 
        completion_notes = p_completion_notes,
        updated_at = NOW()
    WHERE id = p_exchange_id;
    
    -- Transfer time credits
    UPDATE public.profiles 
    SET time_balance = time_balance + v_exchange.time_credits
    WHERE id = v_exchange.provider_id;
    
    UPDATE public.profiles 
    SET time_balance = time_balance - v_exchange.time_credits
    WHERE id = v_exchange.requester_id;
    
    -- Record the transaction
    INSERT INTO public.transactions (
      exchange_id, transaction_type, from_user_id, to_user_id, amount,
      from_balance_before, from_balance_after, to_balance_before, to_balance_after,
      description
    ) VALUES (
      p_exchange_id, 'exchange', v_exchange.requester_id, v_exchange.provider_id, v_exchange.time_credits,
      v_requester_balance, v_requester_balance - v_exchange.time_credits,
      v_provider_balance, v_provider_balance + v_exchange.time_credits,
      'Exchange completed: ' || v_exchange.time_credits || ' time credits'
    );
    
  EXCEPTION
    WHEN OTHERS THEN
      RAISE EXCEPTION 'Failed to complete exchange: %', SQLERRM;
  END;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Function to cancel an exchange
CREATE OR REPLACE FUNCTION cancel_exchange(
  p_exchange_id BIGINT,
  p_reason TEXT DEFAULT NULL
)
RETURNS VOID AS $$
DECLARE
  v_exchange RECORD;
BEGIN
  -- Get the exchange details
  SELECT * INTO v_exchange FROM public.exchanges WHERE id = p_exchange_id;
  
  -- Validate that the current user is involved in the exchange
  IF auth.uid() != v_exchange.provider_id AND auth.uid() != v_exchange.requester_id THEN
    RAISE EXCEPTION 'Only participants can cancel exchanges';
  END IF;
  
  -- Validate the exchange status
  IF v_exchange.status NOT IN ('pending', 'accepted') THEN
    RAISE EXCEPTION 'Only pending or accepted exchanges can be cancelled';
  END IF;
  
  -- Update the exchange status
  UPDATE public.exchanges 
  SET status = 'cancelled', 
      cancelled_at = NOW(), 
      cancellation_reason = p_reason,
      cancelled_by = auth.uid(),
      updated_at = NOW()
  WHERE id = p_exchange_id;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;
```

## Usage Examples

### Create an Exchange Request
```sql
SELECT create_exchange_request(
  offer_id := 1,
  want_id := 2,
  time_credits := 2.5,
  scheduled_date := '2024-01-15 14:00:00+00',
  location := 'Community Center',
  notes := 'Looking forward to learning!'
);
```

### Accept an Exchange Request
```sql
SELECT accept_exchange_request(exchange_id := 1);
```

### Complete an Exchange
```sql
SELECT complete_exchange(
  exchange_id := 1,
  completion_notes := 'Great session, very helpful!'
);
```

## API Endpoints to Create

1. `POST /api/exchanges` - Create exchange request
2. `PUT /api/exchanges/[id]/accept` - Accept exchange request
3. `PUT /api/exchanges/[id]/decline` - Decline exchange request
4. `PUT /api/exchanges/[id]/complete` - Complete exchange
5. `PUT /api/exchanges/[id]/cancel` - Cancel exchange
6. `GET /api/exchanges` - Get user's exchanges
7. `GET /api/transactions` - Get user's transaction history
